    ```javascript
    {
      ticker: 'COST',
      stage: 'pitchDeck',
      state: 'WAVE_2_RUNNING',       // mirrors progress.json state
      startedAt: '2026-03-25T10:00:00Z',
      lastUpdated: '2026-03-25T10:15:00Z',
      elapsedMs: 900000,
      sections: {
        radar: { status: 'complete', startedAt: '...', completedAt: '...', durationMs: 45000, agent: 'business-analyst' },
        simple_predictable: { status: 'complete', startedAt: '...', completedAt: '...', durationMs: 38000, agent: 'business-analyst' },
        market_position: { status: 'running', startedAt: '...', agent: 'competitor-evaluator' },
        barriers_moats: { status: 'pending' },
        // ... all 10 sections
      },
      phases: [
        { phase: 1, status: 'complete', startedAt: '...', completedAt: '...' },
        { phase: 2, status: 'active', startedAt: '...' },
        { phase: 3, status: 'pending' }
      ],
      currentAgent: 'competitor-evaluator',
      completedCount: 2,
      totalSections: 10
    }
    ```

    Export these functions:

    1. `export function initGenerationStatus(ticker, stage)`:
       - Creates initial generation-status.json with all sections as `{ status: 'pending' }`
       - Sets `startedAt` to now, `state` to 'IDLE'
       - Initializes phases array from SECTION_KEYS
       - Writes to disk

    2. `export function updateGenerationStatus(ticker, updates)`:
       - Reads existing generation-status.json from disk
       - Deep-merges `updates` into the status
       - Updates `lastUpdated` to now
       - Computes `elapsedMs` from `startedAt`
       - Computes `completedCount` from sections with status 'complete'
       - Writes back to disk
       - Returns the updated status

    3. `export function startSection(ticker, sectionKey, agent)`:
       - Calls `updateGenerationStatus` with `sections.{sectionKey}.status = 'running'`, `sections.{sectionKey}.startedAt = now`, `sections.{sectionKey}.agent = agent`, `currentAgent = agent`

    4. `export function completeSection(ticker, sectionKey)`:
       - Reads section's `startedAt`, computes `durationMs`
       - Calls `updateGenerationStatus` with `sections.{sectionKey}.status = 'complete'`, `sections.{sectionKey}.completedAt = now`, `sections.{sectionKey}.durationMs = computed`

    5. `export function updatePhaseStatus(ticker, phaseNum, status)`:
       - Updates `phases[phaseNum-1].status` and timestamps

    Use the existing `REPORTS_DIR` and file I/O patterns from progressState.js. The `getStatusPath(ticker)` helper returns `join(REPORTS_DIR, ticker.toUpperCase(), 'generation-status.json')`.

    ```javascript

      updateGenerationStatus(ticker, { state: newState });
    } catch { /* non-critical */ }

    After Step 2 (DataPacket assembly) and before Step 3 (PSR dispatch), add a new **Step 2.5: Pre-Process Filings**:
    ```
    ## Step 2.5: Pre-Process Filings to Markdown

    Convert the most recent 10-K and 10-Q filings to clean markdown BEFORE dispatching PSR agents.

    Run the filing preprocessor:
    ```bash
    node --import scripts/node-esm-loader.js -e "
    import './src/engines/nodeAdapter.js';
    import { fetchFilingMarkdown } from './src/engines/filingMarkdown.js';
    import { extractAllSections } from './src/engines/filingSections.js';
    import { readFileSync, writeFileSync, mkdirSync } from 'fs';
    import { join } from 'path';

    const dp = JSON.parse(readFileSync('.thes1s/reports/{TICKER}/data-packet.json', 'utf8'));
    const filings = dp.filings || [];
    const annuals = filings.filter(f => f.form === '10-K').slice(0, 3);
    const quarterly = filings.filter(f => f.form === '10-Q').slice(0, 4);

    const outDir = '.thes1s/reports/{TICKER}/filings-md';
    mkdirSync(outDir, { recursive: true });

    for (const f of [...annuals, ...quarterly]) {
      const md = await fetchFilingMarkdown(dp.companyInfo.cik, f.accessionNumber, f.primaryDocument);
      const sections = extractAllSections(md);
      const outFile = join(outDir, f.form + '-' + f.filingDate.slice(0, 4) + '.json');
      writeFileSync(outFile, JSON.stringify({ form: f.form, date: f.filingDate, sections, fullLength: md.length }, null, 2));
      console.log('Processed: ' + f.form + ' ' + f.filingDate);
    }
    "
