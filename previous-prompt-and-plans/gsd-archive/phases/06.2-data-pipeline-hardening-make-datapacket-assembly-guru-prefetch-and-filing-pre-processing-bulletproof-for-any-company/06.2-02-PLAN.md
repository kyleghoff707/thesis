       ```javascript
         try {

           if (retry) {
             // Retry once after backoff (per D-09)
             await sleep(backoffMs);
             try {
               return await fn();
             } catch (retryErr) {
               errors.push(`${label}: ${retryErr.message} (after retry)`);
               return null;
             }
           }
           errors.push(`${label}: ${err.message}`);
           return null;
         }
       }
       ```
