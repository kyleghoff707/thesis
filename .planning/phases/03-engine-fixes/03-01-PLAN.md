  ```javascript
  function getFieldValue(data, field, year, isEngine = false) {
    if (!data) return null;
    const names = isEngine ? [resolveFieldName(field), field] : [field];
    for (const name of names) {
      for (const stmtKey of ['income', 'balance', 'cashFlow']) {
        const val = data[stmtKey]?.[year]?.[name];
        if (val != null) return val;
      }
    }
    return null;
  }
  ```
  Note: For engine data, check the ALIAS FIRST (resolveFieldName), then the original name. The alias is the engine's actual field name — e.g., for field `stockholders_equity`, check `equity` first because that's what the engine uses.
