    **Step 3: Create triangulate.mjs** at `validation/scripts/triangulate.mjs`:

    Follow the `compare-morningstar.mjs` pattern for:
    - Shebang: `#!/usr/bin/env node`
    - `.env.local` loading (regex parse for `VITE_FMP_KEY`, `VITE_SIMFIN_KEY`)
    - Path resolution (ROOT, FIXTURES_DIR, REPORTS_DIR, etc.)
    - Browser polyfills (globalThis.localStorage)
    - SEC fetch interceptor with disk cache in `edgar-cache/`
    - Auto-bundle check (`bundled-engines.mjs`)
    - `--ticker` flag parsing (same as compare-morningstar.mjs)
    - Progress reporting to stderr

    Main pipeline:

    ```javascript
    // 1. Load field mapping path
    const FIELD_MAPPING_PATH = path.resolve(FIXTURES_DIR, 'field-mapping.json');
    const FMP_CACHE = path.resolve(ROOT, 'validation/cache/fmp');
    const SIMFIN_CACHE = path.resolve(ROOT, 'validation/cache/simfin');
    const MSTARPY_DATA = path.resolve(ROOT, 'validation/data/mstarpy');

    // 2. Get ticker list (from --ticker flag or all 50 fixture companies)
    const tickers = tickerArg ? tickerArg.split(',') : getFixtureTickers();

    // 3. For each ticker:
    const allCompanyResults = [];
    for (const ticker of tickers) {
      process.stderr.write(`${ticker}... `);

      // 3a. Fetch our XBRL engine data (same as compare-morningstar.mjs)
      const engineData = await fetchEdgarStatements(ticker, { version: 'restated' });

      // 3b. Fetch external source data (parallel where possible)
      const [fmpData, simfinData] = await Promise.all([
        fetchFmpData(ticker, { apiKey: FMP_KEY, cacheDir: FMP_CACHE, fieldMappingPath: FIELD_MAPPING_PATH }),
        fetchSimfinData(ticker, { apiKey: SIMFIN_KEY, cacheDir: SIMFIN_CACHE, fieldMappingPath: FIELD_MAPPING_PATH }),
      ]);
      const mstarpyData = readMstarpyData(ticker, { dataDir: MSTARPY_DATA, fieldMappingPath: FIELD_MAPPING_PATH });

      // 3c. Build source array
      const sources = [
        fmpData && { name: 'fmp', data: fmpData },
        simfinData && { name: 'simfin', data: simfinData },
        mstarpyData && { name: 'mstarpy', data: mstarpyData },
      ].filter(Boolean);

      // 3d. For each canonical field, for each year, classify
      const classifications = [];
      const allYears = collectAllYears(engineData, sources); // union of all year keys
      const allFields = collectAllFields(sources);           // union of all canonical fields

      for (const year of allYears) {
        for (const field of allFields) {
          const thesisValue = getFieldValue(engineData, field, year);
          const sourceValues = sources.map(s => ({
            source: s.name,
            value: getFieldValue(s.data, field, year),
          }));

          const result = classifyField(thesisValue, sourceValues);

          // Root cause tagging for CONSENSUS_DIFF and LIKELY_BUG
          let rootCause = null;
          if (result.classification === 'CONSENSUS_DIFF' || result.classification === 'LIKELY_BUG') {
            const yearContext = {
              prevYearConsensus: getConsensusForYear(sources, field, String(parseInt(year) - 1)),
              nextYearConsensus: getConsensusForYear(sources, field, String(parseInt(year) + 1)),
            };
            rootCause = tagRootCause(thesisValue, result.consensusValue, yearContext);
          }

          classifications.push({
            field,
            year,
            statement: getFieldStatement(field),
            classification: result.classification,
            rootCause,
            thesisValue,
            consensusValue: result.consensusValue,
            sources: Object.fromEntries(sourceValues.map(s => [s.source, s.value])),
          });
        }
      }

      allCompanyResults.push({ ticker, classifications });
      process.stderr.write(`${classifications.filter(c => c.classification === 'MATCH').length} match\n`);
    }

    // 4. Generate reports
    const consoleReport = generateTriangulationConsoleReport(allCompanyResults);
    const fixRecs = generateFixRecommendations(allCompanyResults);

    // 5. Regression diff
    const baselinePath = path.resolve(REPORTS_DIR, 'morningstar-accuracy.json');
    const baseline = JSON.parse(fs.readFileSync(baselinePath, 'utf-8'));
    fixRecs.regressionDiff = generateRegressionDiff(fixRecs, baseline);

    // 6. Write outputs
    fs.writeFileSync(
      path.resolve(REPORTS_DIR, 'fix-recommendations.json'),
      JSON.stringify(fixRecs, null, 2)
    );
    fs.writeFileSync(
      path.resolve(REPORTS_DIR, 'triangulation-report.json'),
      JSON.stringify({ generatedAt: new Date().toISOString(), companies: allCompanyResults }, null, 2)
    );

    // 7. Print console report
    console.log(consoleReport);
    ```

    Helper functions needed in triangulate.mjs:
    - `getFixtureTickers()` -- list all `.json` files in fixtures dir (excluding `field-mapping.json` and `edgar-cache`)
    - `collectAllYears(engineData, sources)` -- union of all year keys across engine + all sources
    - `collectAllFields(sources)` -- union of all canonical field names across all sources
    - `getFieldValue(data, field, year)` -- walk `income[year][field]`, `balance[year][field]`, `cashFlow[year][field]` -- return first non-null
    - `getFieldStatement(field)` -- look up which statement section a field belongs to (from field-mapping.json `_sources` entries)
    - `getConsensusForYear(sources, field, year)` -- collect all source values for a given field+year, return median of agreeing group (for fy_offset detection context)
