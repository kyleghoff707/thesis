    Find the section that handles `progress` (look for the string `'progress'` in the URL path parsing). Add a parallel case:

    ```javascript
    // After the 'progress' case:
    if (parts[1] === 'generation-status') {
      const statusPath = path.join(reportsDir, ticker, 'generation-status.json');
      if (fs.existsSync(statusPath)) {
        const data = fs.readFileSync(statusPath, 'utf8');
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(data);
      } else {
        res.writeHead(404, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ error: 'No generation status found' }));
      }
      return;
    }
    ```

    **2. Extend usePitchDeck hook:**

    Add a `generationStatus` state alongside the existing `report`, `progress`, `loading`, `error` states.

    Add a `fetchGenerationStatus()` function that fetches `/api/thes1s/reports/${ticker}/generation-status`:
    ```javascript
    const [generationStatus, setGenerationStatus] = useState(null);

    async function fetchGenerationStatus() {
      try {
        const res = await fetch(`/api/thes1s/reports/${encodeURIComponent(ticker)}/generation-status`);
        if (cancelled) return null;
        if (res.ok) {
          const data = await res.json();
          if (!cancelled) setGenerationStatus(data);
          return data;
        }
        return null;
      } catch (e) {
        if (!cancelled) console.warn('usePitchDeck: generation status fetch failed:', e.message);
        return null;
      }
    }
    ```

    Modify the polling logic: When `generationStatus` exists and `generationStatus.state !== 'COMPLETE'`, poll `fetchGenerationStatus()` every 2 seconds (alongside the existing progress poll). When `generationStatus.state === 'COMPLETE'`, stop polling and re-fetch the report.

    Update the return value to include `generationStatus`:
    ```javascript
    return { report, progress, generationStatus, loading, error };
    ```

    Also add `fetchGenerationStatus()` to the initial `init()` Promise.all call.
